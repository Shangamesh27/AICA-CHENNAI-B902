function wordDiff(a,b){const x=tokenize(a),y=tokenize(b),out=[];let i=0,j=0;const eq=(u,v)=>norm(u)===norm(v);const skip=v=>$('ignoreSpace').checked&&/^\s+$/.test(v);while(i<x.length||j<y.length){if(i<x.length&&skip(x[i])){if(j<y.length&&skip(y[j])){out.push({type:'equal',a:x[i],b:y[j]});i++;j++;}else{out.push({type:'equal',a:x[i],b:''});i++;}continue;}if(j<y.length&&skip(y[j])){out.push({type:'equal',a:'',b:y[j]});j++;continue;}if(i<x.length&&j<y.length&&eq(x[i],y[j])){out.push({type:'equal',a:x[i],b:y[j]});i++;j++;continue;}let ia=-1,jb=-1;const cap=28;for(let d=1;d<=cap;d++){if(j+d<y.length&&i<x.length&&eq(x[i],y[j+d])){jb=d;break;}if(i+d<x.length&&j<y.length&&eq(x[i+d],y[j])){ia=d;break;}}if(jb>=0){for(let n=0;n<jb;n++)out.push({type:'add',b:y[j++]});}else if(ia>=0){for(let n=0;n<ia;n++)out.push({type:'del',a:x[i++]});}else if(i<x.length&&j<y.length){out.push({type:'del',a:x[i++]});out.push({type:'add',b:y[j++]});}else if(i<x.length)out.push({type:'del',a:x[i++]});else out.push({type:'add',b:y[j++]});}return out;}
function textPart(a,b,label){const ops=wordDiff(a,b);const chunks=[];let group=null,line=1;for(const op of ops){if(op.type==='equal'){if(group){chunks.push(group);group=null;}line+=(op.a.match(/\n/g)||[]).length;continue;}if(!group)group={start:chunks.length,old:'',new:'',line,ops:[]};group.ops.push(op);if(op.a)group.old+=op.a;if(op.b)group.new+=op.b;if(op.a)line+=(op.a.match(/\n/g)||[]).length;}if(group)chunks.push(group);for(const g of chunks){const loc=label+(label.startsWith('Page')||label.startsWith('Slide')?'':', line '+g.line);record(loc,g.old.trim(),g.new.trim(),g.old.trim()&&g.new.trim()?'Modified':g.old.trim()?'Deleted':'Added',{type:'text',part:label});g.id=S.changes.length;for(const op of g.ops)op.id=g.id;}S.chunks.push({label,ops});}
